cmake_minimum_required(VERSION 3.16)
project(soundq_mix_renderer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SQ_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SQ_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
find_package(Threads REQUIRED)
add_library(soundq_mix_object OBJECT src/soundq_mix.cpp)
set_target_properties(soundq_mix_object PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_compile_definitions(soundq_mix_object PRIVATE SQ_BUILD)
target_include_directories(soundq_mix_object PUBLIC include)
add_library(soundq_mix SHARED $<TARGET_OBJECTS:soundq_mix_object>)
target_include_directories(soundq_mix PUBLIC include)
set_target_properties(soundq_mix PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
enable_testing()
add_executable(mix_tests tests/mix_tests.cpp $<TARGET_OBJECTS:soundq_mix_object>)
target_include_directories(mix_tests PRIVATE include)
target_link_libraries(mix_tests PRIVATE Threads::Threads)
add_test(NAME mix_tests COMMAND mix_tests)
add_executable(schedule_tests tests/schedule_tests.cpp $<TARGET_OBJECTS:soundq_mix_object>)
target_include_directories(schedule_tests PRIVATE include)
target_link_libraries(schedule_tests PRIVATE Threads::Threads)
add_test(NAME schedule_tests COMMAND schedule_tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(automation_tests tests/automation_tests.cpp $<TARGET_OBJECTS:soundq_mix_object>)
target_include_directories(automation_tests PRIVATE include)
add_test(NAME automation_wav_pcm COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_automation_wav.py $<TARGET_FILE:automation_tests> ${CMAKE_CURRENT_BINARY_DIR}/automation-proof.wav)
set_tests_properties(automation_wav_pcm PROPERTIES TIMEOUT 60)
add_executable(offline_render examples/offline_render.cpp)
target_link_libraries(offline_render PRIVATE soundq_mix)
add_test(NAME offline_wav_pcm COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_wav.py $<TARGET_FILE:offline_render> ${CMAKE_CURRENT_BINARY_DIR}/offline-proof.wav)
add_executable(c_abi_smoke tests/c_abi_smoke.c)
target_link_libraries(c_abi_smoke PRIVATE soundq_mix)
add_test(NAME c_abi_smoke COMMAND c_abi_smoke)
set_tests_properties(mix_tests schedule_tests offline_wav_pcm c_abi_smoke PROPERTIES TIMEOUT 60)
