cmake_minimum_required(VERSION 3.22)
project(soundq_file_decoder VERSION 0.1.0 LANGUAGES C CXX)
option(SQ_FILE_SANITIZE "ASan/UBSan instrumentation" OFF)
option(SQ_FILE_BUILD_TESTS "Build native test executables" OFF)
option(SQ_FILE_BUILD_SHARED "Build a standalone shared library for worker fixtures" ON)
if(SQ_FILE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  set(SQ_OPUS_SANITIZE ON CACHE BOOL "" FORCE)
endif()
if(NOT TARGET sq_opus_decoder)
  set(SQ_OPUS_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  add_subdirectory(../opus_decoder ${CMAKE_CURRENT_BINARY_DIR}/opus_decoder)
endif()
function(sq_file_configure target)
  target_include_directories(${target} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(${target} PRIVATE sq_opus_decoder)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden)
endfunction()
add_library(sq_file_decoder STATIC src/sq_file_decoder.cpp)
sq_file_configure(sq_file_decoder)
if(SQ_FILE_BUILD_SHARED)
  add_library(sq_file_decoder_shared SHARED src/sq_file_decoder.cpp)
  set_target_properties(sq_file_decoder_shared PROPERTIES OUTPUT_NAME sq_file_decoder)
  sq_file_configure(sq_file_decoder_shared)
endif()
if(SQ_FILE_BUILD_TESTS)
  add_executable(sq_file_tracer tests/tracer.cpp)
  target_link_libraries(sq_file_tracer PRIVATE sq_file_decoder)
  target_compile_options(sq_file_tracer PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  add_executable(sq_file_windows tests/windows.cpp)
  target_link_libraries(sq_file_windows PRIVATE sq_file_decoder)
  target_compile_features(sq_file_windows PRIVATE cxx_std_17)
  target_compile_options(sq_file_windows PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
