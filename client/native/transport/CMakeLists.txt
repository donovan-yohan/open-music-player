cmake_minimum_required(VERSION 3.16)
project(soundq_transport LANGUAGES CXX)

option(SOUNDQ_TRANSPORT_BUILD_TESTS "Build focused portable host tests" ON)
option(SOUNDQ_TEMPO_PCM_TESTS "Build actual decoder/worker/host PCM oracle" OFF)
set(SOUNDQ_TRANSPORT_SANITIZER "" CACHE STRING "Empty, address, or thread")

add_library(soundq_transport STATIC src/soundq_transport.cpp)
target_include_directories(soundq_transport PUBLIC include)
target_compile_features(soundq_transport PUBLIC cxx_std_17)
set_target_properties(soundq_transport PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(soundq_transport PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(SOUNDQ_TEMPO_PCM_TESTS)
    get_filename_component(SQ_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
    set(SQ_ENGINE "${SQ_REPO}/android/app/src/main/cpp/nativeengine")
    set(SQ_FILE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    add_subdirectory(../file_decoder "${CMAKE_CURRENT_BINARY_DIR}/file_decoder")
    include("${CMAKE_CURRENT_SOURCE_DIR}/../third_party/signalsmith/CMakeLists.txt")
    find_package(Threads REQUIRED)
    add_executable(soundq_tempo_pcm_tests tests/tempo_pcm_tests.cpp "${SQ_ENGINE}/callback_host.cpp"
        "${SQ_ENGINE}/worker_deck.cpp" "${SQ_ENGINE}/tempo_processor.cpp")
    target_include_directories(soundq_tempo_pcm_tests PRIVATE "${SQ_ENGINE}")
    set_source_files_properties("${SQ_ENGINE}/tempo_processor.cpp" PROPERTIES COMPILE_OPTIONS "-O2")
    target_compile_options(soundq_tempo_pcm_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(soundq_tempo_pcm_tests PRIVATE soundq_transport sq_file_decoder soundq_signalsmith Threads::Threads)
endif()
if(SOUNDQ_TRANSPORT_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(soundq_live_tempo_tests tests/live_tempo_tests.cpp)
    target_link_libraries(soundq_live_tempo_tests PRIVATE soundq_transport_test_core Threads::Threads)
    target_compile_definitions(soundq_live_tempo_tests PRIVATE SOUNDQ_TRANSPORT_TEST_BARRIERS=1)
    add_test(NAME soundq_live_tempo_handoff COMMAND soundq_live_tempo_tests)
    add_executable(soundq_tempo_tests tests/tempo_tests.cpp)
    target_link_libraries(soundq_tempo_tests PRIVATE soundq_transport)
    add_test(NAME soundq_tempo_map_engine COMMAND soundq_tempo_tests)
    add_executable(soundq_schedule_tests tests/schedule_tests.cpp)
    target_link_libraries(soundq_schedule_tests PRIVATE soundq_transport)
    add_test(NAME soundq_schedule_oracle COMMAND soundq_schedule_tests)
    add_executable(soundq_performance_tests tests/performance_tests.cpp)
    target_link_libraries(soundq_performance_tests PRIVATE soundq_transport)
    add_test(NAME soundq_performance_oracle COMMAND soundq_performance_tests)
    add_executable(soundq_transport_tests tests/transport_tests.cpp)
    target_link_libraries(soundq_transport_tests PRIVATE soundq_transport Threads::Threads)
    add_executable(soundq_residency_tests tests/residency_tests.cpp)
    # Deterministic barriers exist only in this host-test copy, never the library.
    add_library(soundq_transport_test_core STATIC src/soundq_transport.cpp)
    target_include_directories(soundq_transport_test_core PUBLIC include)
    target_compile_features(soundq_transport_test_core PUBLIC cxx_std_17)
    set_target_properties(soundq_transport_test_core PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_definitions(soundq_transport_test_core PRIVATE SOUNDQ_TRANSPORT_TEST_BARRIERS=1)
    target_link_libraries(soundq_residency_tests PRIVATE soundq_transport_test_core Threads::Threads)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(soundq_transport_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_compile_options(soundq_residency_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_compile_options(soundq_transport_test_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(SOUNDQ_TRANSPORT_SANITIZER)
        if(SOUNDQ_TRANSPORT_SANITIZER STREQUAL "address")
            set(transport_sanitizers -fsanitize=address,undefined -fno-omit-frame-pointer)
        elseif(SOUNDQ_TRANSPORT_SANITIZER STREQUAL "thread")
            set(transport_sanitizers -fsanitize=thread -fno-omit-frame-pointer)
        else()
            message(FATAL_ERROR "Unsupported transport sanitizer")
        endif()
        target_compile_options(soundq_live_tempo_tests PRIVATE ${transport_sanitizers})
        target_link_options(soundq_live_tempo_tests PRIVATE ${transport_sanitizers})
        target_compile_options(soundq_tempo_tests PRIVATE ${transport_sanitizers})
        target_link_options(soundq_tempo_tests PRIVATE ${transport_sanitizers})
        target_compile_options(soundq_schedule_tests PRIVATE ${transport_sanitizers})
        target_link_options(soundq_schedule_tests PRIVATE ${transport_sanitizers})
        target_compile_options(soundq_performance_tests PRIVATE ${transport_sanitizers})
        target_link_options(soundq_performance_tests PRIVATE ${transport_sanitizers})
        target_compile_options(soundq_transport PRIVATE ${transport_sanitizers})
        target_compile_options(soundq_transport_tests PRIVATE ${transport_sanitizers})
        target_link_options(soundq_transport_tests PRIVATE ${transport_sanitizers})
        target_compile_options(soundq_transport_test_core PRIVATE ${transport_sanitizers})
        target_compile_options(soundq_residency_tests PRIVATE ${transport_sanitizers})
        target_link_options(soundq_residency_tests PRIVATE ${transport_sanitizers})
    endif()
    add_test(NAME soundq_transport_focused COMMAND soundq_transport_tests)
    add_test(NAME soundq_residency_focused COMMAND soundq_residency_tests)
    set_tests_properties(soundq_transport_focused soundq_residency_focused PROPERTIES TIMEOUT 30)
endif()
