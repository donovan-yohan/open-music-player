cmake_minimum_required(VERSION 3.22)
project(soundq_device_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SQ_DEVICE_TESTS "Build offline host tests (never start an OS device)" ON)
option(SQ_DEVICE_SANITIZE "Enable ASan/UBSan in native host tests" OFF)
if(ANDROID AND ANDROID_PLATFORM_LEVEL LESS 26)
  message(FATAL_ERROR "AAudio host requires Android API 26 or newer")
endif()
if(SQ_DEVICE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(SQ_HOST_WARNINGS -Wall -Wextra -Wpedantic -Werror)
# Source-build pinned target libraries, without propagating host -Werror upstream.
set(SQ_OPUS_BUILD_SHARED OFF CACHE BOOL "Standalone decoder is not packaged" FORCE)
add_subdirectory(../opus_decoder opus_decoder)
function(sq_host_decoder target)
  target_compile_options(${target} PRIVATE ${SQ_HOST_WARNINGS})
  target_link_libraries(${target} PRIVATE sq_opus_decoder)
  # The decoder ABI is looked up from Dart, so the linker sees no C reference.
  # All decoder exports share one C object. -u retains that object on CMake 3.22.
  if(APPLE)
    target_link_options(${target} PRIVATE "-Wl,-u,_sq_opus_decoder_open")
  else()
    target_link_options(${target} PRIVATE "-Wl,-u,sq_opus_decoder_open")
  endif()
endfunction()
find_package(Threads REQUIRED)
set(MIX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../mix_renderer")
set(HOST_COMMON src/device_host.cpp src/device_render.cpp)
# Explicit test-only mode has no OS linkage and is never packaged for devices.
option(SQ_DEVICE_FFI_FAKE_ONLY "Build only the offline Dart FFI fake-platform library" OFF)
if(SQ_DEVICE_FFI_FAKE_ONLY)
  add_library(ffi_actual_core OBJECT "${MIX_ROOT}/src/soundq_mix.cpp")
  target_compile_definitions(ffi_actual_core PRIVATE SQ_BUILD sq_render=sq_ffi_actual_render)
  target_include_directories(ffi_actual_core PRIVATE "${MIX_ROOT}/include")
  set_target_properties(ffi_actual_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
  target_compile_options(ffi_actual_core PRIVATE ${SQ_HOST_WARNINGS})
  add_library(soundq_device_ffi_fake SHARED ${HOST_COMMON} tests/ffi_fake_platform.cpp
    $<TARGET_OBJECTS:ffi_actual_core>)
  target_compile_definitions(soundq_device_ffi_fake PRIVATE SQ_BUILD)
  target_include_directories(soundq_device_ffi_fake PRIVATE include src "${MIX_ROOT}/include")
  target_link_libraries(soundq_device_ffi_fake PRIVATE Threads::Threads)
  sq_host_decoder(soundq_device_ffi_fake)
  return()
endif()
# Compile the existing core translation unit, not its offline test project.
# One shared object exports core + device C ABI; never load two core instances.
add_library(soundq_device SHARED ${HOST_COMMON} "${MIX_ROOT}/src/soundq_mix.cpp")
target_compile_definitions(soundq_device PRIVATE SQ_BUILD)
target_include_directories(soundq_device PUBLIC include "${MIX_ROOT}/include")
target_link_libraries(soundq_device PRIVATE Threads::Threads)
sq_host_decoder(soundq_device)
set_target_properties(soundq_device PROPERTIES CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  target_sources(soundq_device PRIVATE src/device_android.cpp)
  target_link_libraries(soundq_device PRIVATE aaudio)
elseif(APPLE)
  target_sources(soundq_device PRIVATE src/device_apple.cpp)
  target_compile_options(soundq_device PRIVATE -fblocks)
  target_link_libraries(soundq_device PRIVATE "-framework AudioToolbox" "-framework CoreAudio")
else()
  message(FATAL_ERROR "Native device output is implemented only for macOS and Android")
endif()
if(SQ_DEVICE_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(device_invalid_config tests/invalid_config.cpp)
  target_link_libraries(device_invalid_config PRIVATE soundq_device)
  target_compile_options(device_invalid_config PRIVATE ${SQ_HOST_WARNINGS})
  add_test(NAME device_invalid_config COMMAND device_invalid_config)
  add_executable(device_callback tests/render_callback.cpp src/device_render.cpp
    "${MIX_ROOT}/src/soundq_mix.cpp")
  add_executable(device_lifecycle tests/lifecycle.cpp ${HOST_COMMON}
    "${MIX_ROOT}/src/soundq_mix.cpp")
  foreach(target device_callback device_lifecycle)
    target_compile_options(${target} PRIVATE ${SQ_HOST_WARNINGS})
    target_include_directories(${target} PRIVATE include src "${MIX_ROOT}/include")
    target_link_libraries(${target} PRIVATE Threads::Threads)
    add_test(NAME ${target} COMMAND ${target})
  endforeach()
  set_tests_properties(device_invalid_config device_callback device_lifecycle PROPERTIES TIMEOUT 30)
endif()
