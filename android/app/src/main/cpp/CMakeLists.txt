cmake_minimum_required(VERSION 3.22)
project(soundq_android_mix LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(ANDROID)
  # NDK r27 defaults to 4 KiB. Apply both flags to every packaged native target.
  add_link_options("-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
endif()
# AGP invokes Ninja directly. This job pool also fences its builds to one compiler.
set_property(GLOBAL PROPERTY JOB_POOLS soundq_lane=1)
set(CMAKE_JOB_POOL_COMPILE soundq_lane)
set(CMAKE_JOB_POOL_LINK soundq_lane)
if(ANDROID)
  set(SQ_DEVICE_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../../client/native/device_host" device_host)
else()
  # Offline JNI tests need the real renderer/decoder, not an OS output host
  # (or the FFI fake platform, which substitutes render behavior).
  set(SQ_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../client/native")
  set(SQ_OPUS_BUILD_SHARED OFF CACHE BOOL "Standalone decoder is not packaged" FORCE)
  add_subdirectory("${SQ_NATIVE}/opus_decoder" opus_decoder)
  find_package(Threads REQUIRED)
  add_library(soundq_mix_host STATIC "${SQ_NATIVE}/mix_renderer/src/soundq_mix.cpp")
  target_compile_definitions(soundq_mix_host PRIVATE SQ_BUILD)
  target_include_directories(soundq_mix_host PUBLIC "${SQ_NATIVE}/mix_renderer/include")
  target_compile_options(soundq_mix_host PRIVATE -Wall -Wextra -Wpedantic -Werror)
  set_target_properties(soundq_mix_host PROPERTIES CXX_EXTENSIONS OFF
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(soundq_mix_host PUBLIC sq_opus_decoder Threads::Threads)
endif()
add_library(soundq_mix_android SHARED mix_bridge.cpp)
if(ANDROID)
  target_link_libraries(soundq_mix_android PRIVATE soundq_device)
else()
  target_link_libraries(soundq_mix_android PRIVATE soundq_mix_host)
endif()
target_include_directories(soundq_mix_android PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../client/native/opus_decoder/include")
target_compile_options(soundq_mix_android PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(soundq_mix_android PRIVATE ${JNI_INCLUDE_DIRS})
endif()
# Production callback host is independent of the retained inactive legacy mixer.
add_subdirectory(nativeengine)
